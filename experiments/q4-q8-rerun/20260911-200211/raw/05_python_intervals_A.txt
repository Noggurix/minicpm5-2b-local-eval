```
def coalesce_ranges(ranges):
    # Validate input: each start <= end
    for start, end in ranges:
        if start > end:
            raise ValueError(f"Range ({start}, {end}) has start > end")
    # sort by start
    sorted_ranges = sorted(ranges, key=lambda x: (x[0], x[1]))
    if not sorted_ranges:
        return []
    merged = []
    cur_start, cur_end = sorted_ranges[0]
    for start, end in sorted_ranges[1:]:
        # Check if adjacent or overlapping: start <= cur_end + 1
        if start <= cur_end + 1:
            # merge
            cur_end = max(cur_end, end)
        else:
            # finalize current merged interval
            merged.append((cur_start, cur_end))
            cur_start, cur_end = start, end
    # Append the last interval
    merged.append((cur_start, cur_end))
    return merged
```
