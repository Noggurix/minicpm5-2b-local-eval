def coalesce_ranges(ranges):
    # Input validation
    for s, e in ranges:
        if s > e:
            raise ValueError(f"Invalid range: start > end: ({s}, {e})")
    if not ranges:
        return []
    # sort by start then end
    sorted_ranges = sorted(ranges, key=lambda x: (x[0], x[1]))
    merged = []
    cur_start, cur_end = sorted_ranges[0]
    for s, e in sorted_ranges[1:]:
        if s <= cur_end + 1:  # overlapping or touching
            # merge
            cur_end = max(cur_end, e)
        else:
            merged.append((cur_start, cur_end))
            cur_start, cur_end = s, e
    merged.append((cur_start, cur_end))
    return merged
